// server/geo-i18n-middleware.ts — Fastify + Cloudflare CF-IPCountry + Accept-Language
import Fastify, { FastifyRequest } from 'fastify';
import { PrismaClient } from '@prisma/client';

const app = Fastify({ logger: true });
const prisma = new PrismaClient();

interface GeoLocaleConfig {
  country: string;
  lang: string;
  dir: 'ltr' | 'rtl';
  currency: string;
  fxRateFromUsd: number;
  defaultPayoutRail: string;
  availableRails: string[];
}

const REGION_CATALOG: Record<string, GeoLocaleConfig> = {
  BR: { country: 'BR', lang: 'pt-BR', dir: 'ltr', currency: 'BRL', fxRateFromUsd: 5.65, defaultPayoutRail: 'PIX', availableRails: ['PIX', 'MERCADO_PAGO', 'AIRTIME'] },

  MX: { country: 'MX', lang: 'es-419', dir: 'ltr', currency: 'MXN', fxRateFromUsd: 19.8, defaultPayoutRail: 'MERCADO_PAGO', availableRails: ['MERCADO_PAGO', 'SPEI', 'AIRTIME'] },

  IN: { country: 'IN', lang: 'hi-IN', dir: 'ltr', currency: 'INR', fxRateFromUsd: 84.5, defaultPayoutRail: 'UPI', availableRails: ['UPI', 'PAYTM', 'AIRTIME'] },
  DE: { country: 'DE', lang: 'de-DE', dir: 'ltr', currency: 'EUR', fxRateFromUsd: 0.92, defaultPayoutRail: 'SEPA', availableRails: ['SEPA', 'PAYPAL'] },
  US: { country: 'US', lang: 'en-US', dir: 'ltr', currency: 'USD', fxRateFromUsd: 1.0, defaultPayoutRail: 'PAYPAL', availableRails: ['PAYPAL', 'STRIPE_INSTANT'] },
};

function resolveGeoLocale(req: FastifyRequest): GeoLocaleConfig {
  // 1. Edge CDN header (Cloudflare CF-IPCountry or Vercel x-vercel-ip-country)
  const cfCountry = (req.headers['cf-ipcountry'] as string || '').toUpperCase();
  if (cfCountry && REGION_CATALOG[cfCountry]) {
    return REGION_CATALOG[cfCountry];
  }

  // 2. Fallback to Accept-Language header
  const acceptLang = (req.headers['accept-language'] || 'en-US').split(',')[0].trim();
  if (acceptLang.startsWith('pt')) return REGION_CATALOG.BR;
  if (acceptLang.startsWith('es')) return REGION_CATALOG.MX;
  if (acceptLang.startsWith('sw')) return REGION_CATALOG.KE;
  if (acceptLang.startsWith('ar')) return REGION_CATALOG.SA;
  if (acceptLang.startsWith('hi')) return REGION_CATALOG.IN;
  if (acceptLang.startsWith('de')) return REGION_CATALOG.DE;

  return REGION_CATALOG.US;
}

// Endpoint called on app boot to hydrate locale, currency, and localized sponsored feed
app.get('/api/v1/bootstrap', async (req) => {
  const geo = resolveGeoLocale(req);
  const localizedAlerts = await prisma.sponsoredAlert.findMany({
    where: {
      active: true,
      OR: [{ region: geo.country }, { region: 'GLOBAL' }],
    },
    take: 25,
  });

  return {
    geo,
    alerts: localizedAlerts.map((a) => ({
      ...a,
      localRatePerMinute: Number(a.payoutPerMinuteUsd) * geo.fxRateFromUsd,
      displayCurrency: geo.currency,
    })),
  };
});

app.listen({ port: 3000, host: '0.0.0.0' });
// android/app/src/main/java/com/yaytif/app/geo/GeoLocaleManager.kt
package com.yaytif.app.geo

import android.content.Context
import android.telephony.TelephonyManager
import androidx.appcompat.app.AppCompatDelegate
import androidx.core.os.LocaleListCompat
import java.util.Locale
import java.util.TimeZone

data class RegionalProfile(
    val isoCountry: String,
    val languageTag: String,
    val currencyCode: String,
    val primaryPayoutRail: String
)

object GeoLocaleManager {

    fun detectAndApplyLocale(context: Context): RegionalProfile {
        // 1. Detect country via SIM / Network MCC (no GPS permission required)
        val tm = context.getSystemService(Context.TELEPHONY_SERVICE) as? TelephonyManager
        val simCountry = tm?.simCountryIso?.uppercase(Locale.US).orEmpty()
        val netCountry = tm?.networkCountryIso?.uppercase(Locale.US).orEmpty()
        val tzId = TimeZone.getDefault().id

        val resolvedCountry = when {
            simCountry.length == 2 -> simCountry
            netCountry.length == 2 -> netCountry
            tzId.startsWith("America/Sao_Paulo") -> "BR"
            tzId.startsWith("Africa/Nairobi") -> "KE"
            tzId.startsWith("Africa/Lagos") -> "NG"
            tzId.startsWith("Asia/Riyadh") || tzId.startsWith("Africa/Cairo") -> "SA"
            tzId.startsWith("Asia/Kolkata") -> "IN"
            else -> Locale.getDefault().country.ifEmpty { "US" }
        }

        val profile = mapCountryToProfile(resolvedCountry)

        // 2. Apply Android per-app language & automatic RTL layout direction
        val appLocale = LocaleListCompat.forLanguageTags(profile.languageTag)
        AppCompatDelegate.setApplicationLocales(appLocale)

        return profile
    }

    private fun mapCountryToProfile(country: String): RegionalProfile = when (country) {
        "BR" -> RegionalProfile("BR", "pt-BR", "BRL", "PIX")
        "CO", "AR", "MX", "CL", "PE" -> RegionalProfile(country, "es-419", "USD", "MERCADO_PAGO")
        "KE", "TZ", "UG" -> RegionalProfile(country, "sw-KE", "KES", "MPESA")
        "NG" -> RegionalProfile("NG", "en-NG", "NGN", "PAYSTACK")
        "SA", "AE", "EG" -> RegionalProfile(country, "ar-SA", "SAR", "STC_PAY")
        "IN" -> RegionalProfile("IN", "hi-IN", "INR", "UPI")
        "FR", "SN", "CI" -> RegionalProfile(country, "fr-FR", "EUR", "MOBILE_MONEY")
        else -> RegionalProfile(country, "en-US", "USD", "PAYPAL")
    }
}
-- schema.sql — Multi-Currency & Localized Sponsored Offer Storage
CREATE TABLE users (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    phone_hash TEXT UNIQUE NOT NULL,
    iso_country CHAR(2) NOT NULL DEFAULT 'US',
    preferred_lang VARCHAR(12) NOT NULL DEFAULT 'en-US',
    display_currency VARCHAR(6) NOT NULL DEFAULT 'USD',
    wallet_balance_usd NUMERIC(14, 6) NOT NULL DEFAULT 0.000000,
    daily_time_earned_usd NUMERIC(14, 6) NOT NULL DEFAULT 0.000000,
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE TABLE sponsored_alerts (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    advertiser_name TEXT NOT NULL,
    region VARCHAR(8) NOT NULL, -- 'BR', 'KE', 'NG', 'IN', 'GLOBAL'
    category VARCHAR(32) NOT NULL,
    -- JSONB map of localized titles/bodies: {"en-US": {...}, "pt-BR": {...}, "sw-KE": {...}, "ar-SA": {...}}
    localized_copy JSONB NOT NULL,
    payout_per_minute_usd NUMERIC(10, 6) NOT NULL,
    commission_share NUMERIC(4, 2) NOT NULL DEFAULT 0.50,
    affiliate_url_template TEXT NOT NULL,
    budget_remaining_usd NUMERIC(14, 2) NOT NULL,
    active BOOLEAN NOT NULL DEFAULT TRUE
);

CREATE TABLE payout_requests (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID NOT NULL REFERENCES users(id),
    rail VARCHAR(32) NOT NULL, -- 'PIX', 'MERCADO_PAGO', 'MPESA', 'UPI', 'PAYSTACK', 'STC_PAY'
    destination TEXT NOT NULL,
    amount_usd NUMERIC(14, 4) NOT NULL,
    settled_local_amount NUMERIC(14, 2) NOT NULL,
    settled_currency VARCHAR(6) NOT NULL,
    status VARCHAR(16) NOT NULL DEFAULT 'PENDING',
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);
